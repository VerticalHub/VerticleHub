Express middleware:
typescript// server/middleware/auth.ts
export const requireAuth = async (req, res, next) => {
  const session = await validateSession(req.cookies.session_token);
  if (!session) return res.status(401).json({ error: "Unauthorized" });
  req.user = session.user;
  next();
};

export const requireRole = (roles: string[]) => (req, res, next) => {
  if (!roles.includes(req.user.role)) 
    return res.status(403).json({ error: "Forbidden" });
  next();
};

⏰ 3. Lead Aging + Decay
Add to your leads table:
typescriptcapturedAt: timestamp("captured_at").defaultNow(),
agingScore: integer("aging_score").default(100), // degrades over time
Server-side aging calculation (run on a cron or compute on read):
typescript// utils/leadAging.ts
export const computeAgingScore = (capturedAt: Date, baseQuality: number): number => {
  const hoursOld = (Date.now() - capturedAt.getTime()) / 36e5;
  const decayRate = 0.02; // 2% per hour
  const agingMultiplier = Math.max(0.3, 1 - (hoursOld * decayRate));
  return Math.round(baseQuality * agingMultiplier);
};
Add a cron job using node-cron:
bashnpm install node-cron
typescript// server/jobs/ageLeads.ts
import cron from "node-cron";
cron.schedule("*/15 * * * *", async () => {
  const leads = await db.select().from(leadsTable);
  for (const lead of leads) {
    const newScore = computeAgingScore(lead.capturedAt, lead.baseQuality);
    await db.update(leadsTable)
      .set({ agingScore: newScore })
      .where(eq(leadsTable.id, lead.id));
  }
});

app.get("/api/export/leads", requireAuth, async (req, res) => {
  const leads = await db.select().from(leadsTable)
    .where(/* apply filters from req.query */);

  const headers = ["ID","Email","State","Quality","Segment","Captured"];
  const rows = leads.map(l => 
    [l.id, l.email, l.state, l.qualityScore, l.segment, l.capturedAt].join(",")

  res.setHeader("Content-Type", "text/csv");
  res.setHeader("Content-Disposition", `attachment; filename="leads-${Date.now()}.csv"`);
  res.send([headers.join(","), ...rows].join("\n"));
});

📊 8. Conversion Attribution Analytics
New materialized view or query in your analytics route:
typescript// GET /api/analytics/conversion-attribution
const attribution = await db.execute(sql`
  SELECT 
    l.state,
    l.segment,
    l.quality_tier,
    COUNT(*) as total_leads,
    SUM(CASE WHEN la.disposition = 'converted' THEN 1 ELSE 0 END) as conversions,
    ROUND(
      SUM(CASE WHEN la.disposition = 'converted' THEN 1 ELSE 0 END)::decimal 
      / COUNT(*) * 100, 2
    ) as conv_rate
  FROM leads l
  LEFT JOIN lead_activities la ON la.lead_id = l.id
  GROUP BY l.state, l.segment, l.quality_tier
  ORDER BY conv_rate DESC
`);
